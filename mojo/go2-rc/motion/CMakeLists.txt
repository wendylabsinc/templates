cmake_minimum_required(VERSION 3.16)
project(unitree_mojo LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(UNITREE_SDK2_DIR "" CACHE PATH "Path to unitree_sdk2")
if(NOT UNITREE_SDK2_DIR)
  message(FATAL_ERROR "Set UNITREE_SDK2_DIR")
endif()
add_subdirectory("${UNITREE_SDK2_DIR}" "${CMAKE_BINARY_DIR}/unitree_sdk2" EXCLUDE_FROM_ALL)
add_library(unitree_mojo SHARED bridge/src/unitree_mojo.cpp)
target_include_directories(unitree_mojo PUBLIC bridge/include)
target_link_libraries(unitree_mojo PRIVATE unitree_sdk2)
set_target_properties(unitree_mojo PROPERTIES CXX_VISIBILITY_PRESET hidden)
